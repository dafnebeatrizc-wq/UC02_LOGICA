    function drawLabScene() {
      const w = canvas.width;
      const h = canvas.height;
      const targetDino = DINO_DATABASE[currentFossilIndex];

      ctx.clearRect(0, 0, w, h);

      // Background
      const bgGrad = ctx.createLinearGradient(0, 0, 0, h);
      bgGrad.addColorStop(0, '#0F172A');
      bgGrad.addColorStop(1, '#1E293B');
      ctx.fillStyle = bgGrad;
      ctx.fillRect(0, 0, w, h);

      // Tech Grid
      ctx.strokeStyle = '#334155';
      ctx.lineWidth = 1;
      for (let x = 0; x < w; x += 30) {
        ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, h); ctx.stroke();
      }
      for (let y = 0; y < h; y += 30) {
        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(w, y); ctx.stroke();
      }

      // X-Ray Table
      const tableY = h * 0.72;
      ctx.fillStyle = '#38A3F6';
      ctx.fillRect(w * 0.15, tableY, w * 0.7, 12);
      ctx.fillStyle = '#1E3A8A';
      ctx.fillRect(w * 0.18, tableY + 12, w * 0.64, h - tableY);

      // Pillars
      ctx.fillStyle = '#64748B';
      ctx.fillRect(w * 0.15 - 15, tableY - 120, 15, 132);
      ctx.fillRect(w * 0.85, tableY - 120, 15, 132);

      // Indicator LED
      ctx.fillStyle = isAnalyzed ? '#10B981' : '#F59E0B';
      ctx.beginPath(); ctx.arc(w * 0.15 - 7.5, tableY - 110, 5, 0, Math.PI * 2); ctx.fill();

      const centerX = w / 2;
      const centerY = tableY - 45;

      // Draw Fossil State
      if (!isAssembled) {
        drawDisassembledBones(centerX, centerY, targetDino);
      } else {
        if (assemblySuccess) {
          targetDino.drawSkeleton(ctx, centerX, centerY, true);

          ctx.fillStyle = 'rgba(56, 189, 248, 0.2)';
          ctx.beginPath(); ctx.arc(centerX, centerY - 20, 110, 0, Math.PI * 2); ctx.fill();

          ctx.fillStyle = '#38BDF8';
          ctx.font = 'bold 16px Nunito';
          ctx.textAlign = 'center';
          ctx.fillText(`✨ ESQUELETO DE ${targetDino.especie.toUpperCase()} MONTADO! ✨`, centerX, centerY - 110);
        } else {
          drawBrokenSkeleton(centerX, centerY);

          ctx.fillStyle = '#EF4444';
          ctx.font = 'bold 16px Nunito';
          ctx.textAlign = 'center';
          ctx.fillText('❌ ERRO NA MONTAGEM: PARÂMETROS INCORRETOS!', centerX, centerY - 110);
        }
      }

      updateAndDrawParticles();
    }

    function drawDisassembledBones(x, y, dino) {
      ctx.strokeStyle = isAnalyzed ? '#A0C4FF' : '#94A3B8';
      ctx.lineWidth = 3;

      const offsets = [
        { dx: -60, dy: -20, r: 0.3 },
        { dx: -20, dy: -35, r: -0.2 },
        { dx: 30, dy: -10, r: 0.5 },
        { dx: 60, dy: -30, r: -0.4 },
        { dx: 0, dy: 10, r: 0.1 }
      ];

      offsets.forEach((off) => {
        ctx.save();
        ctx.translate(x + off.dx, y + off.dy);
        ctx.rotate(off.r);
        
        ctx.beginPath();
        ctx.moveTo(-15, 0); ctx.lineTo(15, 0);
        ctx.arc(-15, -3, 4, 0, Math.PI * 2);
        ctx.arc(-15, 3, 4, 0, Math.PI * 2);
        ctx.arc(15, -3, 4, 0, Math.PI * 2);
        ctx.arc(15, 3, 4, 0, Math.PI * 2);
        ctx.stroke();

        ctx.restore();
      });

      if (!isAnalyzed) {
        ctx.fillStyle = '#F59E0B';
        ctx.font = 'bold 28px Nunito';
        ctx.textAlign = 'center';
        ctx.fillText('?', x, y - 45);
      }
    }

    function drawBrokenSkeleton(x, y) {
      ctx.strokeStyle = '#EF4444';
      ctx.lineWidth = 4;
      ctx.lineCap = 'round';

      ctx.stroke();

      ctx.fillStyle = '#EF4444';
      ctx.font = 'bold 36px Nunito';
      ctx.textAlign = 'center';
      ctx.fillText('❓', x, y - 50);
    }

    function createConfetti(x, y) {
      const colors = ['#38BDF8', '#C7F9CC', '#FFEC99', '#FFC6FF', '#A0C4FF'];
      for (let i = 0; i < 40; i++) {
        particles.push({
          x: x, y: y,
          vx: (Math.random() - 0.5) * 8,
          vy: (Math.random() - 0.7) * 8,
          color: colors[Math.floor(Math.random() * colors.length)],
          size: Math.random() * 6 + 3,
          life: 1.0
        });
      }
    }

    function updateAndDrawParticles() {
      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx; p.y += p.vy;
        p.vy += 0.2;
        p.life -= 0.02;

        if (p.life <= 0) {
          particles.splice(i, 1);
          continue;
        }

        ctx.fillStyle = p.color;
        ctx.globalAlpha = p.life;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
        ctx.globalAlpha = 1.0;
      }
    }

    function showToast(message, type = 'info') {
      toastMsg.textContent = message;
      canvasToast.className = "absolute bottom-4 left-1/2 transform -translate-x-1/2 px-4 py-2 rounded-xl text-xs sm:text-sm font-bold shadow-lg transition-all duration-300 opacity-100 translate-y-0 z-20 flex items-center gap-2 ";

      if (type === 'success') {
        canvasToast.classList.add('bg-emerald-500', 'text-white');
        toastIcon.className = "fa-solid fa-circle-check";
      } else if (type === 'error') {
        canvasToast.classList.add('bg-rose-500', 'text-white');
        toastIcon.className = "fa-solid fa-triangle-exclamation";
      } else {
        canvasToast.classList.add('bg-babyBlue-600', 'text-white');
        toastIcon.className = "fa-solid fa-magnifying-glass";
      }

      setTimeout(() => {
        canvasToast.classList.add('opacity-0', 'translate-y-4');
        canvasToast.classList.remove('opacity-100', 'translate-y-0');
      }, 3500);
    }

    function analyzeFossil() {
      if (isAnalyzed) {
        showToast("Este fóssil já foi analisado! Confira o caderno.", "info");
        return;
      }

      const dino = DINO_DATABASE[currentFossilIndex];
      
      scannerLine.classList.remove('hidden');
      let scanCount = 0;
      const scanInterval = setInterval(() => {
        scanCount += 5;
        scannerLine.style.top = `${(scanCount % 100)}%`;
        if (scanCount > 200) {
          clearInterval(scanInterval);
          scannerLine.classList.add('hidden');
          completeAnalysis(dino);
        }
      }, 30);
    }

      // Show location banner
      locationBanner.classList.remove('hidden');
      locationText.textContent = dino.mensagemDescoberta;

      // Populate Notebook

      showToast("Análise concluída! Origem geográfica e características reveladas.", "success");

    function executeMontarEsqueleto() {
      if (!isAnalyzed) {
        showToast("Analise o fóssil com a Lupa antes de montar!", "error");
        return;
      }

      const dino = DINO_DATABASE[currentFossilIndex];
      const valDentes = selectDentes.value;
      const valPostura = selectPostura.value;
      const valTamanho = selectTamanho.value;
      const valEspecie = selectEspecie.value;

      const isDentesCorrect = valDentes === dino.dentes;
      const isPosturaCorrect = valPostura === dino.postura;
      const isTamanhoCorrect = valTamanho === dino.tamanho;
      const isEspecieCorrect = valEspecie === dino.especie;

      isAssembled = true;

      if (isDentesCorrect && isPosturaCorrect && isTamanhoCorrect && isEspecieCorrect) {
        assemblySuccess = true;
        score = Math.min(6, score + 1);
        updateScoreUI();
        createConfetti(canvas.width / 2, canvas.height / 2);
        showToast(`Sucesso! montar_esqueleto("${dino.especie}") reconstruiu o fóssil!`, "success");
      } else {
        assemblySuccess = false;
        let errorReason = [];
        if (!isDentesCorrect) errorReason.push("dentes");
        if (!isPosturaCorrect) errorReason.push("postura");
        if (!isTamanhoCorrect) errorReason.push("tamanho");
        if (!isEspecieCorrect) errorReason.push("espécie");

        showToast(`Erro na função! Variável(is) incorreta(s): ${errorReason.join(', ')}.`, "error");
      }

      drawLabScene();
    }
